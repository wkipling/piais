# Append to /boot/firmware/config.txt (Raspberry Pi OS) or /boot/config.txt
# (older images / Alpine).  Do not replace the whole file.

# PiCAN 2: MCP2515 on SPI0, 16 MHz crystal, interrupt GPIO25.
dtparam=spi=on
dtoverlay=mcp2515-can0,oscillator=16000000,interrupt=25

# If your PiCAN 2 has a different crystal, change oscillator= to match the
# marking on the can.  8 MHz and 16 MHz are the common values.

# Optional second controller on a PiCAN 2 Duo.  Check the board silkscreen.
